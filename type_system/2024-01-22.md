### Strange behaviours

```ocaml
module Alias(X: sig type 'a key  type ('a, 'b) value end) = struct
  type 'a key = 'a X.key
  type ('a, 'b) value = ('a, 'b) X.value
end

module M = struct
  type 'a key
  type ('a, 'b) value = 'b
  let print: 'a. 'a key -> unit = fun _ -> assert false
end

module Aliased = Alias(M)
open Aliased
type 'b polyiter = { f : 'a. 'a key -> ('a, 'b) value -> unit }
type g = { g:'a. 'a key -> int -> unit}

let conv {g} =
  { f = g }
```
```
Error: This field value has type
         'b Aliased.key ->
         ('b, ('b, int) Aliased.value) Aliased.value -> unit
       which is less general than
         'a. 'a Aliased.key -> ('a, 'c) Aliased.value -> unit
```

Simpler:

```ocaml
module Mk(X: sig type 'x t end) = struct
  type 'a alias = 'a X.t
  type any = { f: 'x. 'x alias }
end
module X = struct
  type 'x t  = int
end
module M = Mk(X)
open M
let ok (f: int ) = { f }
let fail f = { f }
```
